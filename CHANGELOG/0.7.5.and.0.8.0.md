## ✅ 2. Limite de sécurité 10 agents / 10 instances avec prompt interactif
```python
# train/cli.py — post_process_args()
SAFETY_CAP = 10

if not nolimit and (args.num_agents > SAFETY_CAP or args.num_instances > SAFETY_CAP):
    print("  SAFETY LIMIT REACHED")
    print(f"  Requested: {args.num_agents} agent(s), {args.num_instances} instance(s)")
    print("  Launching too many instances can freeze your PC...")
    answer = input("  Proceed with high values? [y/N]: ").strip().lower()
    if answer != 'y':
        args.num_agents    = min(args.num_agents, SAFETY_CAP)
        args.num_instances = min(args.num_instances, SAFETY_CAP)
# --nolimit pour bypass (sans prompt)
```

## ✅ 4. `capture_mixin.py` — Thread capture asynchrone isolé
```python
# environment/capture_mixin.py
class CaptureMixin:
    def _async_capture_loop(self):
        """
        Thread daemon : capture → preprocess → push queue.
        Rate limited à ~30 FPS.
        Logs drop rate seulement si > 10%.
        """
        while self._capture_running:
            frame  = self.frame_capture.capture_frame()
            visual = self.preprocessor.process_and_stack_numpy(frame)
            try:
                self._obs_queue.put(visual, block=False)
                self._frames_captured += 1
            except queue.Full:
                self._frames_dropped += 1
            time.sleep(1.0 / 30)
```

```python
# environment/reward_bridge_mixin.py
class RewardBridgeMixin:
    def _calculate_reward(self, action) -> Tuple[float, dict]:
        """
        Lit le game state, détecte les dégâts, délègue au RewardCalculator,
        enrichit step_info avec les stats GUI.
        """
        took_damage = self._detect_damage(raw_memory)
        reward = self.reward_calc.calculate(
            self.prev_raw_memory, raw_memory, action, info,
            took_damage=took_damage
        )
        self._enrich_info(info, raw_memory)
        return reward, info
```
