## ✅ 3. `_EnvContextFilter` — Routing des logs par (agent, env)

```python
# info/advanced_logging.py
class _EnvContextFilter(logging.Filter):
    """
    Filtre les records pour ne garder que ceux de (agent_id, env_id) spécifique.
    Utilisé sur console.log dans env_N/ pour que chaque fichier
    ne reçoive que ses propres logs.
    """
    def __init__(self, agent_id: int, env_id: int):
        self.agent_id = agent_id
        self.env_id   = env_id

    def filter(self, record) -> bool:
        current_agent = AgentContext.get_current_agent()
        current_env   = EnvContext.get_current_env()
        # Si pas de contexte → seulement agent 0 / env 0 (logs de démarrage)
        if current_agent is None and current_env is None:
            return self.agent_id == 0 and self.env_id == 0
        agent_ok = (current_agent == self.agent_id) if current_agent else (self.agent_id == 0)
        if current_env is None:
            return agent_ok and self.env_id == 0
        return agent_ok and (current_env == self.env_id)
```

## ✅ 4. `AgentContextFilter` — Pour errors.log (niveau agent)

```python
# info/advanced_logging.py
class AgentContextFilter(logging.Filter):
    """Filtre par agent_id uniquement (errors.log partagé entre envs d'un agent)."""
    def filter(self, record) -> bool:
        current = AgentContext.get_current_agent()
        if current is None:
            return self.agent_id == 0  # Startup logs → agent 0
        return current == self.agent_id
```

## ✅ 6. `build_loggers()` — Création une instance par env

```python
# train/logging_setup.py
def build_loggers(args, allocation_result, session_timestamp):
    """Crée un TrainingLogger par (agent, instance) pair."""
    for env_idx in range(args.num_instances):
        aid = inst_to_agent.get(env_idx, 0)
        tl = TrainingLogger(
            experiment_name=args.name,
            agent_id=aid,
            instance_id=env_idx,
            num_agents=args.num_agents,
        )
        loggers_by_env[env_idx] = tl
    return list(loggers_by_env.values()), loggers_by_env[0]
```

## ✅ 7. `LoggingCallback.record_step()` — Routing vers le bon logger env

```python
# info/advanced_logging.py
def record_step(self, reward, action, done, info, env_idx=0):
    """Appelé depuis MultiAgentTrainer avec env_idx explicite."""
    EnvContext.set_current_env(env_idx)  # Pour _EnvContextFilter

    _env_logger = (
        self.all_env_loggers[env_idx]
        if env_idx < len(self.all_env_loggers)
        else self.training_logger
    )
    # → Écrit dans le bon training_data.jsonl de cet env
    _orig = self.training_logger
    self.training_logger = _env_logger
    self._process_step(reward=reward, action=action, done=done, info=info)
    self.training_logger = _orig
```

## ✅ 8. `_CompressedRotatingFileHandler` — Rotation avec compression gzip

```python
# info/advanced_logging.py — Les vieux logs sont compressés automatiquement
class _CompressedRotatingFileHandler(RotatingFileHandler):
    def doRollover(self) -> None:
        super().doRollover()
        for i in range(1, self.backupCount + 1):
            rolled = f"{self.baseFilename}.{i}"
            if os.path.exists(rolled):
                with open(rolled, 'rb') as f_in, gzip.open(f"{rolled}.gz", 'wb') as f_out:
                    shutil.copyfileobj(f_in, f_out)
                os.remove(rolled)
# Tailles :
#   console.log      : 5 MB, 3 backups (.1.gz, .2.gz, .3.gz)
#   errors.log       : 2 MB, 2 backups
#   reward_debug.log : 20 MB, 5 backups
```
